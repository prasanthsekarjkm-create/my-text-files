CREATE OR REPLACE PROCEDURE process_employee_salaries 
IS
   TYPE emp_id_table IS TABLE OF employees.employee_id%TYPE;
   emp_ids emp_id_table; -- Collection variable

   CURSOR emp_cursor IS SELECT employee_id FROM employees
   WHERE department_id = 10;

   errors NUMBER; -- To track the number of errors
   err_msg VARCHAR2(4000); -- To store error messages

BEGIN
   OPEN emp_cursor;
   FETCH emp_cursor BULK COLLECT INTO emp_ids;
   CLOSE emp_cursor;

   IF emp_ids.COUNT > 0 THEN
      FORALL i IN emp_ids.FIRST .. emp_ids.LAST SAVE EXCEPTIONS
         UPDATE employees SET salary = salary * 1.10
         WHERE employee_id = emp_ids(i);
   ELSE
      DBMS_OUTPUT.PUT_LINE('No employees found for the given criteria.');
   END IF;

EXCEPTION
   WHEN OTHERS THEN
      errors := SQL%BULK_EXCEPTIONS.COUNT;
      DBMS_OUTPUT.PUT_LINE('Number of errors: ' || errors);

      FOR i IN 1 .. errors 
      LOOP
         err_msg := 'Error at index ' || SQL%BULK_EXCEPTIONS(i).ERROR_INDEX ||
                    ': ' || SQLERRM(SQL%BULK_EXCEPTIONS(i).ERROR_CODE);
         DBMS_OUTPUT.PUT_LINE(err_msg);
      END LOOP;
END;

CREATE OR REPLACE PROCEDURE Update_Employee_Status
IS
    CURSOR emp_cursor IS SELECT emp_id, salary FROM employees;
    v_emp_id   employees.emp_id%TYPE;
    v_salary   employees.salary%TYPE;
BEGIN
    OPEN emp_cursor;
    LOOP
        FETCH emp_cursor INTO v_emp_id, v_salary;
        EXIT WHEN emp_cursor%NOTFOUND;
        IF v_salary > 100000 THEN
            UPDATE employees SET status = 'Good' WHERE emp_id = v_emp_id;
        ELSE
            UPDATE employees SET status = 'Well' WHERE emp_id = v_emp_id;
       END IF;
    END LOOP;
    CLOSE emp_cursor;
    COMMIT;
    DBMS_OUTPUT.PUT_LINE('Employee statuses updated successfully.');
EXCEPTION
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('An error occurred: ' || SQLERRM);
        ROLLBACK; -- Rollback in case of an error
        -- Ensure cursor is closed in case of an exceptionk
        IF emp_cursor%ISOPEN THEN
            CLOSE emp_cursor;
        END IF;
END Update_Employee_Status;
/

DECLARE
   name_to_reverse VARCHAR2(100) := 'Prasanth'; -- Replace with the name you want to reverse
   reversed_name   VARCHAR2(100) := '';
BEGIN
   -- Loop through each character of the string in reverse order
   FOR i IN REVERSE 1..LENGTH(name_to_reverse) LOOP
      reversed_name := reversed_name || SUBSTR(name_to_reverse, i, 1);
   END LOOP;

   -- Print the reversed name
   DBMS_OUTPUT.PUT_LINE('Original Name: ' || name_to_reverse);
   DBMS_OUTPUT.PUT_LINE('Reversed Name: ' || reversed_name);
END;

DECLARE
   -- Declare an associative array type for employee names
   TYPE emp_name_array IS TABLE OF VARCHAR2(100) INDEX BY PLS_INTEGER;
   
   -- Declare an associative array variable
   emp_names emp_name_array;

   -- Variable for the array index
   idx PLS_INTEGER := 0;
BEGIN
   -- Populate the associative array with employee names from the emp table
   FOR rec IN (SELECT empname FROM emp) LOOP
      idx := idx + 1; -- Increment the index
      emp_names(idx) := rec.empname; -- Store employee name in the array
   END LOOP;

   -- Print the employee names stored in the associative array
   idx := emp_names.FIRST; -- Get the first index
   WHILE idx IS NOT NULL LOOP
      DBMS_OUTPUT.PUT_LINE('Employee Name: ' || emp_names(idx)); -- Print the name
      idx := emp_names.NEXT(idx); -- Move to the next index
   END LOOP;
END;

DECLARE
   CURSOR emp_cursor IS SELECT empid, empname, salary, deptid FROM emp;
   v_empid emp.empid%TYPE;
   v_empname emp.empname%TYPE;
   v_salary emp.salary%TYPE;
   v_deptid emp.deptid%TYPE;
   salary_threshold CONSTANT NUMBER := 30000;

BEGIN
   OPEN emp_cursor;
   LOOP
      FETCH emp_cursor INTO v_empid, v_empname, v_salary, v_deptid;
      EXIT WHEN emp_cursor%NOTFOUND;
      IF v_salary < salary_threshold THEN
         DBMS_OUTPUT.PUT_LINE('Validation Failed: ' || v_empname || ' has a low salary (' || v_salary || ').');
      ELSIF v_deptid IS NULL THEN
         DBMS_OUTPUT.PUT_LINE('Validation Failed: ' || v_empname || ' has no department ID.');
      ELSE
         DBMS_OUTPUT.PUT_LINE('Validation Passed: ' || v_empname || ' is valid.');
      END IF;
   END LOOP;
   CLOSE emp_cursor;
END;

IS

  FOR rec IN (SELECT * FROM (SELECT * FROM customer ORDER BY customer_id) WHERE ROWNUM <= 5)
  LOOP
    IF rec.customer_id IS NULL THEN

    ELSE
      DBMS_OUTPUT.PUT_LINE('customer_id: ' || rec.customer_id || 
                           ' name: ' || rec.name || 
                           ' active: ' || rec.is_active);
    END IF;
  END LOOP;

EXCEPTION
  WHEN OTHERS THEN
    DBMS_OUTPUT.PUT_LINE('An error occurred: ' || SQLERRM);
